# Every Android permission the installed app may request, one per line.
# The build fails if the APK asks for anything not listed here, so a new
# permission (for example internet access) can never sneak in unnoticed.
#
# Added automatically by AndroidX so the app's own broadcast receivers stay private:
io.github.secnewt.dialer.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION
io.github.secnewt.dialer.debug.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION

# Caller announcement (off by default; asked for only when turned on):
# know when a call rings, read the caller's number, and look up the contact name.
android.permission.READ_PHONE_STATE
android.permission.READ_CALL_LOG
android.permission.READ_CONTACTS
#
# Favorites (asked for only when Favorites or Contacts is opened):
# star and unstar contacts in the phone's own contacts, which Do Not Disturb uses.
android.permission.WRITE_CONTACTS
